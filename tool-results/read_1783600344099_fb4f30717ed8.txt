   363→function expand(raw: (number | string)[][]): Star[] {
   364→  return raw.map((r) => {
   365→    const raH = r[5] as number;
   366→    const decD = r[6] as number;
   367→    return {
   368→      hip: r[0] as number,
   369→      hd: r[1] as number,
   370→      name: r[2] as string,
   371→      bayer: r[3] as string,
   372→      flamsteed: r[4] as number,
   373→      raHours: raH,
   374→      decDeg: decD,
   375→      mag: r[7] as number,
   376→      spectral: r[8] as string,
   377→      distanceLy: r[9] as number,
   378→      pmRa: r[10] as number,
   379→      pmDec: r[11] as number,
   380→      constellation: r[12] as string,
   381→      ra: raH * HRS2RAD,
   382→      dec: decD * DEG2RAD,
   383→    } as Star;
   384→  });
   385→}

   387→/** Deduplicate by HIP number — keep the first occurrence. */
   388→function dedupe(stars: Star[]): Star[] {
   389→  const seen = new Set<number>();
   390→  const out: Star[] = [];
   391→  for (const s of stars) {
   392→    const key = Math.round(s.hip);
   393→    if (seen.has(key)) continue;
   394→    seen.add(key);
   395→    out.push(s);
   396→  }
   397→  return out;
   398→}

   413→/** Map spectral class letter to a display color (RGB) for rendering. */
   414→export function spectralColor(spectral: string): [number, number, number] {
   415→  const c = spectral.charAt(0).toUpperCase();
   416→  switch (c) {
   417→    case "O": return [155, 176, 255];
   418→    case "B": return [170, 191, 255];
   419→    case "A": return [202, 215, 255];
   420→    case "F": return [248, 247, 255];
   421→    case "G": return [255, 244, 209];
   422→    case "K": return [255, 210, 161];
   423→    case "M": return [255, 174, 132];
   424→    default:  return [255, 255, 255];
   425→  }
   426→}
